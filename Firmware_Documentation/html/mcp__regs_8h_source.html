74/*
75 * ---- IOCON_SEQOP: the one that matters for us -------------------------
76 *
77 * SEQOP = 0 (POR): after each byte the address pointer INCREMENTS. Normal
78 * behaviour. A 2-byte read from 0x12 returns GPIOA, GPIOB and leaves the
79 * pointer at 0x14.
80 *
81 * SEQOP = 1: the pointer does not increment - EXCEPT that at BANK = 0 it
82 * TOGGLES between the two halves of an A/B pair. Park it on 0x12 and a
83 * 2-byte read returns GPIOA, GPIOB and leaves it back on 0x12.
84 *
85 * That is what lets our scan read be address-byte + 2 data bytes with no
86 * register pointer write: 29 bit periods instead of 48, a 39% saving on
87 * every encoder read.
88 *
89 * THE TRAP: the pointer toggles PER BYTE, not per transaction. Read an odd
90 * number of bytes and it is left sitting on GPIOB, and every subsequent
91 * read returns the wrong port. All scan reads must be even-length.
92 *
93 * Several widely-copied blog posts state that byte mode is SEQOP = 0. They
94 * are wrong; the datasheet bit definition is explicit that 1 disables
95 * incrementing.
96 */
97

106/*
107 * There is no register pointer. The first byte after the address byte is
108 * an opcode, decoded from its top bits:
109 *
110 * C2 C1 C0 W1 W0
111 * 0 0 x Fast Write (2 bytes per channel)
112 * 0 1 0 0 0 Multi-Write (input register only, 3 B/ch)
113 * 0 1 0 1 0 Sequential Write (input reg + EEPROM, from ch N)
114 * 0 1 0 1 1 Single Write (input reg + EEPROM, one ch)
115 * 0 1 1 Write Address Bits (needs LDAC timing, see below)
116 * 1 0 0 Write VREF (1 byte)
117 * 1 0 1 Write Power-Down (2 bytes)
118 * 1 1 0 Write Gain (1 byte)
119 */
120
121#define DAC_CMD_FAST 0x00 /* bits 7-6 = 00 */
122#define DAC_CMD_MULTIWRITE 0x40 /* 0100 0000 | (ch << 1) | udac */
123#define DAC_CMD_SEQWRITE 0x50 /* 0101 0000 | (ch << 1) | udac */
124#define DAC_CMD_SINGLEWRITE 0x58 /* 0101 1000 | (ch << 1) | udac */
125#define DAC_CMD_ADDRBITS 0x60 /* 0110 0001 / 0010 / 0011 sequence */
126#define DAC_CMD_WRITE_VREF 0x80 /* 1000 VA VB VC VD */
127#define DAC_CMD_WRITE_PD 0xA0 /* 1010 pd_a pd_b, then pd_c pd_d 0000 */
128#define DAC_CMD_WRITE_GAIN 0xC0 /* 1100 GA GB GC GD */
129
130/*
131 * ---- Fast Write frame (what our scheduler streams) --------------------
132 *
133 * [addr<<1|0]
134 * [0 0 PD1 PD0 D11 D10 D9 D8] \ channel A
135 * [D7 D6 D5 D4 D3 D2 D1 D0] /
136 * ... repeated for B, C, D (may stop early after any channel)
137 *
138 * 8 data bytes for all four channels. 81 bit periods = 210 us at 400 kHz.
139 *
140 * CRITICAL: Fast Write carries NO VREF and NO GAIN bits. Those come from
141 * whatever is in EEPROM. Configure them ONCE at init with Write VREF and
142 * Write Gain (or a Sequential Write) before relying on Fast Write, or the
143 * output scaling will be whatever the factory or a previous build left.
144 */
145
146#define DAC_FAST_HI(v12) ((uint8_t)(((v12) >> 8) & 0x0F)) /* PD = 00 */
147#define DAC_FAST_LO(v12) ((uint8_t)((v12) & 0xFF))
148
149/*
150 * ---- Multi-Write frame (single channel, when you need VREF/gain) ------
151 *
152 * [addr<<1|0]
153 * [0 1 0 0 0 DAC1 DAC0 UDAC]
154 * [VREF PD1 PD0 Gx D11 D10 D9 D8]
155 * [D7..D0]
156 *
157 * 36 bit periods = 90 us. Cheaper than Fast Write when only one channel
158 * changed, and it is the only per-transfer way to set VREF/gain.
159 *
160 * UDAC = 0 means the output updates immediately (equivalent to pulsing
161 * LDAC). UDAC = 1 defers until LDAC goes low - use that to update several
162 * channels simultaneously.
163 */
164
